<script>
"use strict";

/*
  EthiopiaAI
  Developer: Eyu Bura

  IMPORTANT:
  To enable real AI answers, deploy a backend endpoint
  at /api/chat. The backend must accept:
      POST { "message": "...", "history": [...] }

  It should return:
      { "reply": "..." }

  Never place a private AI API key in this HTML file.
*/

const chat = document.getElementById("chat");
const input = document.getElementById("question");
const sendBtn = document.getElementById("sendBtn");
const micBtn = document.getElementById("micBtn");
const speakBtn = document.getElementById("speakBtn");
const stopBtn = document.getElementById("stopBtn");
const statusBox = document.getElementById("status");

let voiceEnabled = true;
let recognition = null;
let listening = false;
let busy = false;

const history = [];

function setStatus(text) {
  statusBox.textContent = text;
}

function addMessage(text, role) {
  const item = document.createElement("div");
  item.className = "message " + role;
  item.textContent = text;
  chat.appendChild(item);
  chat.scrollTop = chat.scrollHeight;
  return item;
}

function speak(text) {
  if (!voiceEnabled || !("speechSynthesis" in window)) {
    return;
  }

  window.speechSynthesis.cancel();

  const utterance = new SpeechSynthesisUtterance(text);

  // Browser/phone must have a suitable Amharic voice
  // installed for natural Amharic speech.
  utterance.lang = /[\u1200-\u137F]/.test(text)
    ? "am-ET"
    : "en-US";

  utterance.rate = 0.92;
  utterance.pitch = 1;

  window.speechSynthesis.speak(utterance);
}

function localHelp(message) {
  const q = message.trim().toLowerCase();

  if (
    q === "ሰላም" ||
    q === "ሰላም ነህ" ||
    q === "hello" ||
    q === "hi"
  ) {
    return "ሰላም! 😊 EthiopiaAI ነኝ። ዛሬ ምን ልርዳህ?";
  }

  if (
    q.includes("ማን ሰራህ") ||
    q.includes("ማን ፈጠረህ") ||
    q.includes("who created you") ||
    q.includes("who made you")
  ) {
    return "እኔ EthiopiaAI ነኝ። በEyu Bura የተሰራ የAI ፕሮጀክት ነኝ።";
  }

  if (
    q.includes("ስምህ") ||
    q.includes("your name")
  ) {
    return "ስሜ EthiopiaAI ነው። በEyu Bura የተሰራ የAI ፕሮጀክት ነኝ።";
  }

  return null;
}

async function askAI(message) {
  /*
    This endpoint does not exist until you deploy
    and configure your own backend.

    The backend securely connects to an AI model.
  */

  const response = await fetch("/api/chat", {
    method: "POST",
    headers: {
      "Content-Type": "application/json"
    },
    body: JSON.stringify({
      message: message,
      history: history.slice(-12),
      language: /[\u1200-\u137F]/.test(message)
        ? "Amharic"
        : "Auto"
    })
  });

  if (!response.ok) {
    throw new Error("AI server error: " + response.status);
  }

  const data = await response.json();

  if (typeof data.reply !== "string" || !data.reply.trim()) {
    throw new Error("The AI server returned no answer.");
  }

  return data.reply.trim();
}

async function sendMessage() {
  if (busy) return;

  const message = input.value.trim();
  if (!message) return;

  input.value = "";
  addMessage(message, "user");

  const localAnswer = localHelp(message);

  if (localAnswer) {
    addMessage(localAnswer, "ai");
    history.push(
      { role: "user", content: message },
      { role: "assistant", content: localAnswer }
    );
    speak(localAnswer);
    return;
  }

  busy = true;
  sendBtn.disabled = true;
  setStatus("AI እያሰበ ነው...");

  const loading = addMessage("እባክህ ጠብቅ...", "ai");

  try {
    const answer = await askAI(message);
    loading.textContent = answer;

    history.push(
      { role: "user", content: message },
      { role: "assistant", content: answer }
    );

    // Keep conversation context at a manageable size.
    if (history.length > 20) {
      history.splice(0, history.length - 20);
    }

    speak(answer);
    setStatus("");
  } catch (error) {
    loading.textContent =
      "እውነተኛ AI አሁን አልተገናኘም። " +
      "የAI backend ማዘጋጀትና /api/chat ማገናኘት ያስፈልጋል። " +
      "የኢንተርኔትህን ግንኙነትም አረጋግጥ።";

    setStatus("AI server አልተገናኘም።");
    console.error(error);
  } finally {
    busy = false;
    sendBtn.disabled = false;
  }
}

sendBtn.addEventListener("click", sendMessage);

input.addEventListener("keydown", function(event) {
  if (event.key === "Enter") {
    sendMessage();
  }
});

speakBtn.addEventListener("click", function() {
  voiceEnabled = true;
  setStatus("የመልስ ድምፅ በርቷል።");
});

stopBtn.addEventListener("click", function() {
  if ("speechSynthesis" in window) {
    window.speechSynthesis.cancel();
  }
  voiceEnabled = false;
  setStatus("ድምፅ ቆሟል።");
});

const SpeechRecognition =
  window.SpeechRecognition ||
  window.webkitSpeechRecognition;

if (SpeechRecognition) {
  recognition = new SpeechRecognition();
  recognition.lang = "am-ET";
  recognition.continuous = false;
  recognition.interimResults = false;
  recognition.maxAlternatives = 1;

  recognition.onstart = function() {
    listening = true;
    micBtn.textContent = "🎙️ እያዳመጥኩ ነው...";
    setStatus("አሁን ተናገር።");
  };

  recognition.onresult = function(event) {
    const transcript =
      event.results[0][0].transcript;

    input.value = transcript;
    setStatus("የተሰማው፦ " + transcript);
    sendMessage();
  };

  recognition.onerror = function(event) {
    setStatus(
      "የድምፅ ስህተት፦ " + event.error +
      ". የMicrophone ፈቃድን አረጋግጥ።"
    );
  };

  recognition.onend = function() {
    listening = false;
    micBtn.textContent = "🎤 በድምፅ ጠይቀኝ";
  };

  micBtn.addEventListener("click", function() {
    if (listening) {
      recognition.stop();
      return;
    }

    try {
      recognition.start();
    } catch (error) {
      setStatus("የድምፅ ማዳመጥ አልተጀመረም። እንደገና ሞክር።");
    }
  });
} else {
  micBtn.disabled = true;
  micBtn.textContent = "የድምፅ ማዳመጥ በዚህ Browser አይደገፍም";
}

</script>
